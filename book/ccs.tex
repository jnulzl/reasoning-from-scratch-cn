% -----------------------------------------------------------------------------
%  【补丁】minted v3 × tcolorbox × subfiles 路径冲突
% -----------------------------------------------------------------------------
%  现象（TeX Live 2026 更新 minted 3.8 / tcolorbox 6.10 之后出现）：
%      ! Package minted Error: Cannot locate file
%        "book/content/main/main.listing" (kpsewhich failed).
%      ! FancyVerb Error: No verbatim file book/content/main/main.listing
%      → 所有代码块（465 处）全部编译失败，退化为报错而不是高亮。
%
%  原因：tcolorbox 的 tcblisting 会把代码正文写到输出目录下的临时文件
%      （默认 \kvtcb@listingfile = \jobname.listing，即 ./main.listing），
%      然后调用 \inputminted{...}{main.listing}。
%      而 minted 的 \minted@define@inputfilepath 会无条件地把当前
%      \import@path（subfiles/import 设为当前子文件所在目录
%      "book/content/main/"）前缀到该文件名上，于是去找
%      "book/content/main/main.listing" —— 这个文件根本不存在。
%
%  修法：保留 minted「优先按子文件目录解析」的原意，只在
%      \import@path + 文件名 确实不存在时，退回按输出目录（当前目录）解析。
%      对 \inputminted 的常规用法无影响。
\makeatletter
\def\minted@define@inputfilepath#1{%
  \IfPackageLoadedTF{import}%
   {\IfFileExists{\import@path#1}%
     {\edef\minted@inputfilepath{\import@path#1}}%
     {\edef\minted@inputfilepath{#1}}}%
   {\edef\minted@inputfilepath{#1}}}
\makeatother

% -----------------------------------------------------------------------------
%  章节命令（对应模板的 \myChapter / \myPart；本书没有「部分」，并增加前言/附录两类）
% -----------------------------------------------------------------------------
\ExplSyntaxOn
% 编号正文章节：\myChapter{章号}{标题}{文件} —— 章首有巨型浅灰数字与钢蓝横线
% 标题里可用 \\ 手工断行（避免长章名折出「孤字」）；断行符在目录与页眉里会被替换成空格
\tl_new:N \l__my_toctitle_tl
\NewDocumentCommand { \myChapter } { mmm }
{
  \setcounter{chapter}{#1}
  \setcounter{section}{0}
  \mychapterset{#1}%
  \tl_set:Nn \l__my_toctitle_tl { #2 }
  \tl_replace_all:Nnn \l__my_toctitle_tl { \\ } { \space }
  \chapmark{第~#1~章\quad \l__my_toctitle_tl}%  \chapter* 不触发 \chaptermark，必须自己写标记
  \chapter*{#2}
  \mychapterclear
  \addcontentsline{toc}{chapter}{第~#1~章\quad \l__my_toctitle_tl}
  \subfile{#3}
}
% 附录：\myAppendix{序号 1..7}{标题}{文件}（自动显示为「附录 A/B/C…」）
\NewDocumentCommand { \myAppendix } { mmm }
{
  \setcounter{chapter}{#1}
  \setcounter{section}{0}
  \mychapterclear
  \chapter*{附录~\Alph{chapter}\quad #2}
  \addcontentsline{toc}{chapter}{附录~\Alph{chapter}\quad #2}
  \chapmark{附录~\Alph{chapter}\quad #2}
  \subfile{#3}
}
% 前言类：进目录但无编号
\NewDocumentCommand { \myFront } { mm }
{
  \mychapterclear
  \chapter*{#1}
  \addcontentsline{toc}{chapter}{#1}
  \chapmark{#1}
  \subfile{#2}
}
% 扉页/版权类：不进目录
\NewDocumentCommand { \myFrontNoToc } { m }
{
  \clearpage
  \markboth{}{}
  \subfile{#1}
}
\ExplSyntaxOff
