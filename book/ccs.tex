% -----------------------------------------------------------------------------
%  【关键】minted v3 在 Windows 下的缓存目录必须是绝对路径
% -----------------------------------------------------------------------------
%  latexminted 写任何文件前都要过 latexrestricted 的安全校验
%  （_restricted_pathlib.py 的 writable_dir()）：
%      elif self.is_absolute() and not any(self.is_relative_to(p)
%                                          for p in self.tex_texmfoutput_roots()):
%          -> 拒绝
%  也就是说：
%    * 绝对路径  → 必须落在 tex_texmfoutput_roots()（由 TEXMFOUTPUT /
%      TEXMF_OUTPUT_DIRECTORY 环境变量填充，且是绝对路径）之内；
%    * 相对路径  → 直接判负（is_absolute() 为假，短路进不了任何放行分支）。
%  而 minted 默认 \minted@cachedir = "_minted"，生成的文件名是
%      "_minted/<md5>.highlight.minted"、"<md5>.style.minted"、"<md5>.pygtex" …
%  全是【相对路径】——于是 465 处代码块全部写入失败。
%  更坑的是 latexminted 把真实原因吞进 _<md5>.message.minted，
%  只在 LaTeX 侧留下极具误导性的一句：
%      Cannot write file "_xxx.index.minted" outside working directory,
%      TEXMFOUTPUT, and TEXMF_OUTPUT_DIRECTORY.
%  （看到这句先别去查工作目录，几乎一定是相对/绝对路径判定问题）
%
%  实测（latexminted 0.7.0 + latexrestricted 0.6.2）：
%      cachedir="_minted"（默认）        → 写入被拒
%      cachedir=""                       → 写入被拒（仍是相对路径）
%      cachedir="<项目根绝对路径>"        → 正常 ← 唯一可行
%
%  修法：把 \minted@cachedir 设成【项目根目录的绝对路径】。
%  连带效果：\minted@cachepath 变成 "<绝对路径>/"，于是 latexminted 拿到的
%  所有待写路径都是绝对路径，全部落在项目根目录内、校验通过。
%
%  项目根目录怎么求？——读 TEXMF_OUTPUT_DIRECTORY / TEXMFOUTPUT 环境变量
%  （由 build.bat 用 %~dp0 设成项目根的绝对路径）。
%  ⚠ 不要用 `kpsewhich --var-value PWD`！PWD 是 bash 专有的环境变量，
%    在 cmd.exe / PowerShell 下并不存在，kpsewhich 会返回【空】，
%    于是 \minted@cachedir 变成空串 → 仍旧全灭。这个坑很隐蔽：
%    在 git-bash 里测试全部通过，一双击 build.bat 就挂。
%  读回来后必须：
%    1) 判空 —— 若两个变量都没设，宁可保持 minted 默认值并给出警告，
%       也不要把它设成空串（空串会连带炸出 \pydatawritekeyedefvalue 等一堆错误）；
%    2) 剥掉 CRLF 残留（Windows 下 kpsewhich 输出带 \r）。
%  注意：必须直接改 \minted@cachedir 宏本身，不能用 \setminted{cachedir=}，
%  因为 minted 只为 cachedir 注册了带 .estore 的描述子，空值会被 pgfkeys 拒绝。
\makeatletter
% 1) 依次尝试两个环境变量，取第一个非空者
\CatchFileDef{\mintedProjectRoot}{|"kpsewhich --var-value TEXMF_OUTPUT_DIRECTORY"}%
  {\minted@standardcatcodes\endlinechar=-1}%
\def\minted@stripcr#1\r{#1}%
\edef\mintedProjectRoot{\expandafter\minted@stripcr\mintedProjectRoot\r}%
\ifx\mintedProjectRoot\@empty
  \CatchFileDef{\mintedProjectRoot}{|"kpsewhich --var-value TEXMFOUTPUT"}%
    {\minted@standardcatcodes\endlinechar=-1}%
  \edef\mintedProjectRoot{\expandafter\minted@stripcr\mintedProjectRoot\r}%
\fi
% 2) 只在非空时才覆盖 minted 的默认值；为空则保留 "_minted" 并明确报警
\ifx\mintedProjectRoot\@empty
  \PackageWarning{minted-cn}{%
    Cannot determine project root: neither TEXMF_OUTPUT_DIRECTORY nor
    TEXMFOUTPUT is set. Code blocks will FAIL to highlight.^^J%
    Please build via build.bat, or set TEXMF_OUTPUT_DIRECTORY to the^^J%
    absolute path of the project root before running xelatex.}%
\else
  \edef\minted@cachedir{\mintedProjectRoot}%
  \typeout{minted-cn: cachedir = \minted@cachedir}%
\fi
\makeatother

% -----------------------------------------------------------------------------
%  【补丁】minted v3 × tcolorbox × subfiles 路径冲突
% -----------------------------------------------------------------------------
%  现象（TeX Live 2026 更新 minted 3.8 / tcolorbox 6.10 之后出现）：
%      ! Package minted Error: Cannot locate file
%        "book/content/main/main.listing" (kpsewhich failed).
%      ! FancyVerb Error: No verbatim file book/content/main/main.listing
%      → 所有代码块（465 处）全部编译失败，退化为报错而不是高亮。
%
%  原因：tcolorbox 的 tcblisting 会把代码正文写到输出目录下的临时文件
%      （默认 \kvtcb@listingfile = \jobname.listing，即 ./main.listing），
%      然后调用 \inputminted{...}{main.listing}。
%      而 minted 的 \minted@define@inputfilepath 会无条件地把当前
%      \import@path（subfiles/import 设为当前子文件所在目录
%      "book/content/main/"）前缀到该文件名上，于是去找
%      "book/content/main/main.listing" —— 这个文件根本不存在。
%
%  修法：保留 minted「优先按子文件目录解析」的原意，只在
%      \import@path + 文件名 确实不存在时，退回按输出目录（当前目录）解析。
%      对 \inputminted 的常规用法无影响。
\makeatletter
\def\minted@define@inputfilepath#1{%
  \IfPackageLoadedTF{import}%
   {\IfFileExists{\import@path#1}%
     {\edef\minted@inputfilepath{\import@path#1}}%
     {\edef\minted@inputfilepath{#1}}}%
   {\edef\minted@inputfilepath{#1}}}
\makeatother

% ---------------------------------------------------------------------------
%  章标题的「上间距」（= titlesec 的 before-skip，即标题块距版心顶的距离）
%  ★ 想调整 \myFront（前言/致谢/索引…）标题的高低，只改 \myFrontTopSkip 这一个值 ★
%    换算关系（实测 1:1）：标题末行基线 y = 181.55 + (本值 − 10.89) bp
%      · 每 +1bp，标题下移 1bp；每 −1bp，标题上移 1bp
%      · 想直接落在 y = Y：写 Y − 170.66 bp（允许负值；上移别越过页眉基线 33.30）
%    默认 10.89bp → 基线 181.55，与章首页标题完全对齐。
%  注意 \myChapter / \myAppendix / \myFront / 目录 共用同一个 numberless 章格式，
%  而 \myFront 用「局部覆盖 + 用完还原」的方式单独取值，所以改 \myFrontTopSkip
%  不会影响章首页与附录。
%  ⚠ \myChapterTopSkip 若改动，章首页的钢蓝横线（y=193.80）与巨型章号（基线
%    217.88）是绝对坐标 overlay，不会跟着动，必须同步改 \chapterdecor 里的 yshift。
%  ── 改了值却「没反应」的三种常见原因 ──
%   1. 看错页：**目录页不吃 \myFrontTopSkip**（\tableofcontents 走的是 \myChapterTopSkip），
%      章首页、附录同理。\myFrontTopSkip 只作用于 index.tex 里的 \myFront 调用：
%      献词 / 前言 / 致谢 / 关于本书 / 关于作者 / 关于封面插图 / 索引。
%   2. 改得太少：换算 1:1，10.89→15 只差 4bp（约 1.4mm），肉眼基本看不出。
%      要验证请一次跨 30–50bp，或用下面 \myFront 里的 \typeout 到 main.log 里核对实际取值。
%   3. 没编译到：请确认改的是 latex_source_v2/book/ccs.tex，且编译根文件是本目录的 main.tex。
% ---------------------------------------------------------------------------
\newcommand{\myChapterTopSkip}{10.89bp}
\newcommand{\myFrontTopSkip}{-10.89bp}   % ★ 想调整 \myFront（前言/致谢/索引…）标题高低，改这个值
\newcommand{\chapterdecor}{%
  \ifx\mychaptergiant\empty\else      % 与 \empty 比较「含义」，无需展开，空则整块不画

      \node[anchor=base east, inner sep=0bp, outer sep=0bp]
        at ([xshift=-37.18bp,yshift=-217.88bp]current page.north east)
        {\fontsize{225bp}{225bp}\selectfont\itshape\color{chapgray}\mychaptergiant};
      \draw[chapterblue, line width=0.5bp]
        ([xshift=-57.18bp,yshift=-193.80bp]current page.north east) -- ++(-300bp,0);
    \end{tikzpicture}%
  \fi}

\ExplSyntaxOn
% 编号正文章节：\myChapter{章号}{标题}{文件} —— 章首有巨型浅灰数字与钢蓝横线
% 标题里可用 \\ 手工断行（避免长章名折出「孤字」）；断行符在目录与页眉里会被替换成空格
\tl_new:N \l__my_toctitle_tl
\NewDocumentCommand { \myChapter } { mmm }
{
  \setcounter{chapter}{#1}
  \setcounter{section}{0}
  \mychapterset{#1}%
  \tl_set:Nn \l__my_toctitle_tl { #2 }
  \tl_replace_all:Nnn \l__my_toctitle_tl { \\ } { \space }
  \chapmark{第~#1~章\quad \l__my_toctitle_tl}%  \chapter* 不触发 \chaptermark，必须自己写标记
  \chapter*{#2}
  \mychapterclear
  \addcontentsline{toc}{chapter}{第~#1~章\quad \l__my_toctitle_tl}
  \subfile{#3}
}
% 附录：\myAppendix{序号 1..7}{标题}{文件}（自动显示为「附录 A/B/C…」）
\NewDocumentCommand { \myAppendix } { mmm }
{
  \setcounter{chapter}{#1}
  \setcounter{section}{0}
  \mychapterclear
  \chapter*{附录~\Alph{chapter}\quad #2}
  \addcontentsline{toc}{chapter}{附录~\Alph{chapter}\quad #2}
  \chapmark{附录~\Alph{chapter}\quad #2}
  \subfile{#3}
}
% 前言类：进目录但无编号
% 标题上间距用 \myFrontTopSkip 单独指定（想调高低只改那个值），
% 用完立刻还原成章标题默认值，避免影响其后的 \myChapter / \myAppendix。
\NewDocumentCommand { \myFront } { mm }
{
  \mychapterclear
  \typeout{[myFront]TopSkip=\myFrontTopSkip}%   ← 诊断：该值会打进 main.log
                                               %   改完在 main.log 里搜 "myFront]TopSkip" 即可确认是否生效
  \titlespacing*{name=\chapter,numberless}{0bp}{\myFrontTopSkip}{2.6em}
  \chapter*{#1}
  \titlespacing*{name=\chapter,numberless}{0bp}{\myChapterTopSkip}{2.6em}
  \addcontentsline{toc}{chapter}{#1}
  \chapmark{#1}
  \subfile{#2}
}
